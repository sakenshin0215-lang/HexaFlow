HexaFlow V1.0/
├── hexaflow/                        # 核心引擎源码包
│   ├── core/                        # 核心驱动层
│   │   ├── engine.py                # 工作流执行引擎 (DAG Runner)
│   │   ├── state_machine.py         # 全局状态机与断点续传管理
│   │   ├── exceptions.py            # 自定义异常 (如 DomMismatchError, NeedsHumanIntervention)
│   │   └── trace_recorder.py        # 操作轨迹录制器
│   ├── browser/                     # 浏览器与物理隔离层
│   │   ├── context_manager.py       # 多账号隔离管理器 (基于 Playwright Context)
│   │   ├── sandbox.py               # 动作执行沙盒 (封装 click, type, scroll 等安全操作)
│   │   └── stealth.py               # 反爬与拟人化特征注入
│   ├── agents/                      # 智能体大脑层
│   │   ├── planner.py               # 根据自然语言生成初始 JSON 蓝图
│   │   ├── healer.py                # 异常分析与自我修复专家
│   │   └── prompts/                 # 核心系统提示词管理
│   ├── tools/                       # 能力工具箱 (供硬代码或 Agent 调用)
│   │   ├── dom_parser.py            # 快速 DOM 树裁剪与关键信息提取
│   │   ├── vision_utils.py          # 预留的截图与图像预处理接口
│   │   └── interactions.py          # 复杂的组合交互 (如处理特定 Web3 钱包授权弹窗)
│   └── memory/                      # 存储与上下文层
│       ├── local_db.py              # 本地数据库交互 (存储状态、日志、账号元数据)
│       └── workspace/               # 本地工作区
│           ├── traces/              # 存放录制好的黄金轨迹 (JSON/YAML)
│           └── screenshots/         # 存放报错时的现场快照
├── tests/                           # 单元测试与集成测试
├── config.yaml                      # 全局配置文件 (并发数、模型 API Key 等)
├── main.py                          # 命令行入口 (CLI)
└── requirements.txt                 # 依赖清单
